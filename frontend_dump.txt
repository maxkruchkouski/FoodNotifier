import { useState, useEffect, useRef } from 'react'

export default function CustomerBoard({ restaurantId }) {
    const [cookingOrders, setCookingOrders] = useState([])
    const [readyOrders, setReadyOrders] = useState([])
    const [error, setError] = useState(null)
    const prevReadyIds = useRef([])

    const fetchBoardOrders = async () => {
        try {
            const response = await fetch(`/api/v1/board/orders?restaurant_id=${restaurantId}`)
            if (!response.ok) {
                if (response.status === 403) throw new Error('billing_blocked')
                throw new Error('Ошибка сервера')
            }
            const orders = await response.json()
            setCookingOrders(orders.filter(o => o.status === 'cooking'))
            const ready = orders.filter(o => o.status === 'ready')
            setReadyOrders(ready)
            setError(null)

            const currentReadyIds = ready.map(o => o.order_number)
            const hasNewReady = currentReadyIds.some(num => !prevReadyIds.current.includes(num))
            if (hasNewReady && prevReadyIds.current.length > 0) playSound()
            prevReadyIds.current = currentReadyIds
        } catch (err) {
            setError(err.message === 'billing_blocked' ? 'Доступ заблокирован. Оплатите тариф.' : 'Соединение потеряно...')
        }
    }

    const playSound = () => {
        try {
            const ctx = new (window.AudioContext || window.webkitAudioContext)()
            const osc = ctx.createOscillator()
            const gain = ctx.createGain()
            osc.type = 'sine'
            osc.frequency.setValueAtTime(587.33, ctx.currentTime)
            osc.frequency.setValueAtTime(880, ctx.currentTime + 0.15)
            gain.gain.setValueAtTime(0.1, ctx.currentTime)
            gain.gain.exponentialRampToValueAtTime(0.01, ctx.currentTime + 0.4)
            osc.connect(gain)
            gain.connect(ctx.destination)
            osc.start()
            osc.stop(ctx.currentTime + 0.4)
        } catch (e) {}
    }

    useEffect(() => {
        if (!restaurantId) return
        fetchBoardOrders()
        const interval = setInterval(fetchBoardOrders, 5000)
        return () => clearInterval(interval)
    }, [restaurantId])

    if (error) return <div style={{ background: '#000', color: '#e74c3c', height: '100vh', display: 'flex', justifyContent: 'center', alignItems: 'center', fontSize: '24px' }}>⚠️ {error}</div>

    return (
        <div style={{ background: '#0a0a0c', color: '#fff', height: '100vh', width: '100vw', display: 'flex', overflow: 'hidden', fontFamily: 'sans-serif' }}>
            <div style={{ width: '50%', borderRight: '4px solid #1c1c24', display: 'flex', flexDirection: 'column' }}>
                <div style={{ background: '#13131a', padding: '30px', textAlign: 'center', fontSize: '32px', fontWeight: 'bold', color: '#8e929b' }}>ГОТОВЯТСЯ</div>
                <div style={{ padding: '40px', display: 'grid', gridTemplateColumns: 'repeat(auto-fill, minmax(140px, 1fr))', gap: '25px' }}>
                    {cookingOrders.map((o, i) => <div key={i} style={{ background: '#16161e', borderRadius: '16px', padding: '20px', textAlign: 'center', fontSize: '36px', fontWeight: 'bold', color: '#a5a7b0' }}>{o.order_number}</div>)}
                </div>
            </div>
            <div style={{ width: '50%', display: 'flex', flexDirection: 'column', background: '#050c05' }}>
                <div style={{ background: '#1c3d1c', padding: '30px', textAlign: 'center', fontSize: '32px', fontWeight: 'bold', color: '#2ecc71' }}>ГОТОВО!</div>
                <div style={{ padding: '40px', display: 'grid', gridTemplateColumns: 'repeat(auto-fill, minmax(180px, 1fr))', gap: '30px' }}>
                    {readyOrders.map((o, i) => <div key={i} style={{ background: '#2ecc71', borderRadius: '20px', padding: '25px', textAlign: 'center', fontSize: '54px', fontWeight: '#000', color: '#000' }}>{o.order_number}</div>)}
                </div>
            </div>
        </div>
    )
}

import { useState, useEffect, useRef } from 'react'

export default function KitchenMonitor({ restaurantId }) {
    const [orders, setOrders] = useState([])
    const [error, setError] = useState(null)
    const intervalRef = useRef(null)

    const loadOrders = async () => {
        try {
            const res = await fetch(`/api/v1/kitchen/orders?restaurant_id=${restaurantId}`)
            if (!res.ok) throw new Error(res.status === 403 ? 'blocked' : 'err')
            setOrders(await res.json())
            setError(null)
        } catch (err) {
            setError(err.message === 'blocked' ? '🔒 Срок действия лицензии FoodNotifier истек.' : 'Потеря соединения...')
        }
    }

    useEffect(() => {
        loadOrders()
        intervalRef.current = setInterval(loadOrders, 3000)
        return () => clearInterval(intervalRef.current)
    }, [restaurantId])

    const action = async (id, type) => {
        await fetch(`/api/v1/kitchen/orders/${id}/${type}?restaurant_id=${restaurantId}`, { method: 'PATCH' })
        loadOrders()
    }

    if (error) return <div style={{ background: '#0c0c0e', color: '#e74c3c', padding: '40px', textAlign: 'center', fontFamily: 'sans-serif' }}><h2>{error}</h2></div>

    return (
        <div style={{ background: '#141419', color: '#fff', minHeight: '100vh', padding: '20px', fontFamily: 'sans-serif' }}>

                        <h1 style={{ color: '#ff9f1c', margin: '5px 0' }}>#{o.order_number}</h1>
                        <button onClick={() => action(o.id, o.status === 'pending' ? 'start' : 'ready')} style={{ width: '100%', padding: '10px', background: o.status === 'pending' ? import { useState, useEffect, useRef } from 'react'

export default function GuestTracker() {
    const [restaurantId, setRestaurantId] = useState('default_cafe')
    const [orderNumber, setOrderNumber] = useState('')
    const [currentOrderId, setCurrentOrderId] = useState(null)
    const [screen, setScreen] = useState('form') // form, status
    const [orderStatus, setOrderStatus] = useState('pending')
    const checkInterval = useRef(null)

    useEffect(() => {
        const params = new URLSearchParams(window.location.search)
        const resId = params.get('restaurant') || 'default_cafe'
        setRestaurantId(resId)

        fetch('/api/v1/analytics/track', {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ restaurant_id: resId })
        }).catch(() => {})

        const savedId = localStorage.getItem(`active_order_id_${resId}`)
        const savedNum = localStorage.getItem(`active_order_num_${resId}`)
        if (savedId && savedNum) {
            setCurrentOrderId(savedId)
            setOrderNumber(savedNum)
            setScreen('status')
        }
        return () => clearInterval(checkInterval.current)
    }, [])

    useEffect(() => {
        if (screen === 'status' && currentOrderId) {
            const fetchStatus = async () => {
                const res = await fetch(`/api/v1/orders/${currentOrderId}/status?restaurant_id=${restaurantId}`)
                const data = await res.json()
                if (data.status) setOrderStatus(data.status)
            }
            fetchStatus()
            checkInterval.current = setInterval(fetchStatus, 4000)
        } else {
            clearInterval(checkInterval.current)
        }
        return () => clearInterval(checkInterval.current)
    }, [screen, currentOrderId])

    const handleStartTracking = async (e) => {
        e.preventDefault()
        const res = await fetch(`/api/v1/orders?restaurant_id=${restaurantId}`, {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ order_number: orderNumber })
        })
        const data = await res.json()
        if (data.success) {
            setCurrentOrderId(data.order_id)
            localStorage.setItem(`active_order_id_${restaurantId}`, data.order_id)
            localStorage.setItem(`active_order_num_${restaurantId}`, orderNumber)
            setScreen('status')
        }
    }

    return (
        <div style={{ background: '#121212', color: '#fff', minHeight: '100vh', display: 'flex', justifyContent: 'center', alignItems: 'center', fontFamily: 'sans-serif' }}>
            <div style={{ background: '#1e1e1e', padding: '30px 24px', borderRadius: '20px', width: '340px', textAlign: 'center', border: '1px solid #2d2d2d' }}>
                <h1 style={{ fontSize: '24px', margin: '0 0 10px 0', color: '#ff9f1c' }}>FoodNotifier</h1>
                <p style={{ color: '#a0a0a0', fontSize: '12px' }}>Заведение: {restaurantId}</p>

                {screen === 'form' ? (
                    <form onSubmit={handleStartTracking}>
                        <input type="number" value={orderNumber} onChange={e => setOrderNumber(e.target.value)} placeholder="Номер заказа" style={{ width: '100%', padding: '14px', background: '#2a2a2a', border: '1px solid #3d3d3d', borderRadius: '8px', color: '#fff', textAlign: 'center', fontSize: '20px', margin: '20px 0', boxSizing: 'border-box' }} required />
                        <button type="submit" style={{ width: '100%', padding: '14px', background: '#ff9f1c', border: 'none', borderRadius: '8px', fontWeight: 'bold', cursor: 'pointer' }}>Следить</button>
                    </form>
                ) : (
                    <div>
                        <div style={{ background: '#2a2a2a', padding: '6px 12px', borderRadius: '20px', display: 'inline-block', margin: '15px 0' }}>ЗАКАЗ № {orderNumber}</div>
                        <h2 style={{ color: orderStatus === 'ready' ? '#2ecc71' : '#ff9f1c', margin: '20px 0' }}>
                            {orderStatus === 'ready' ? '🎉 Готов!' : orderStatus === 'cooking' ? '🔥 Готовится...' : '⏳ В очереди'}
                        </h2>
                        <button onClick={() => { localStorage.clear(); setScreen('form'); }} style={{ background: 'none', border: '1px solid #3d3d3d', color: '#a0a0a0', padding: '8px', width: '100%', borderRadius: '8px', cursor: 'pointer' }}>Сбросить</button>
                    </div>
                )}
            </div>
        </div>
    )
}

    const [selectedSlug, setSelectedSlug] = useState('subway')
    const [data, setData] = useState(null)
    const [period, setPeriod] = useState('week')
    const [newName, setNewName] = useState('')
    const [newSlug, setNewSlug] = useState('')

    useEffect(() => {
        const params = new URLSearchParams(window.location.search)
        if (params.get('view')) setView(params.get('view'))
        if (params.get('restaurant_id')) setSelectedSlug(params.get('restaurant_id'))
        if (params.get('restaurant')) setSelectedSlug(params.get('restaurant'))
    }, [])

    const loadData = async () => {
        try {
            const res = await fetch(`/api/v1/dashboard?stats=${selectedSlug}&period=${period}`)
            if (!res.ok) return
            const json = await res.json()
            setData(json)
        } catch (err) {
            console.error("API Error:", err)
        }
    }

    useEffect(() => {
        if (view === 'admin') loadData()
    }, [selectedSlug, period, view])

    const handleCreate = async (e) => {
        e.preventDefault()
        if (!newName || !newSlug) return
        try {
            const res = await fetch('/api/v1/restaurants', {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({ name: newName, slug: newSlug })
            })
            if (res.ok) { setNewName(''); setNewSlug(''); loadData(); }
        } catch (e) {}
    }

    const handleToggle = async (id) => {
        await fetch(`/api/v1/restaurants/${id}/toggle`, { method: 'PATCH' })
        loadData()
    }

    const handleDelete = async (id) => {
        if (confirm("Удалить заведение?")) {
            await fetch(`/api/v1/restaurants/${id}`, { method: 'DELETE' })
            loadData()
        }
    }

    if (view === 'board') return <CustomerBoard restaurantId={selectedSlug} />
    if (view === 'track') return <GuestTracker />

                Загрузка экосистемы FoodNotifier...

    const restaurants = data.restaurants || []

                    <button onClick={() => setView('kitchen')} style={{ background: '#34495e', color: '#fff', border: 'none', padding: '10px 15px', borderRadius: '6px', cursor: 'pointer' }}>👨‍🍳 На кухню</button>
